# Runtime dependencies for ResearchMesh-Router.
#   pip install -r requirements.txt
#
# The first four are the router proper. Everything under "local tools" backs the
# tool modules in core/, which execute work on this machine, so the router needs
# those packages plus the apt packages (libreoffice, pandoc, python3-tk, scrot,
# tesseract-ocr, xvfb, pulseaudio-utils) and `playwright install chromium`.
#
# Every package here is required. Each is imported inside its tool's `execute`,
# so a missing one breaks that tool (with an install hint), not the whole app.

anthropic>=1,<2          # the ceiling stops a fresh install drifting past a breaking major;
                         # raise it only after ruff, mypy, smoke_test.py and a live call pass
mcp[cli]>=2.0,<3         # MCP client + Streamable HTTP transport
# 2.0 is a breaking major and this code uses its API: `streamable_http_client`
# in mcp_client.py (headers arrive via an httpx2 client, not a `headers=`
# argument), `read_timeout_seconds` as a plain float on ClientSession, and
# snake_case model fields in core/tools.py (`input_schema`, `is_error`,
# `mime_type`). None of it runs on 1.x. The camelCase spellings survive as
# serialization aliases, so building a model works either way while attribute
# reads break: a 1.x/2.x mismatch in core/tools.py is a runtime AttributeError
# that no import check catches, which is why `mypy .` is a gate. The `<3`
# matches claude-agent-sdk, so the two can share a venv.
prompt-toolkit>=3.0.51   # interactive CLI REPL
pydantic>=2.0            # imported directly in mcp_client.py; also required by mcp/anthropic

# --- local tools ---------------------------------------------------------
# Backings for the tool modules in core/.
playwright>=1.40.0       # `browser_*` — headless browser (core/browser.py)
# These two floors are where CurveZMQ transport encryption became available:
# 8.9.1 is KernelManager's `transport_encryption` trait (8.9.0 shipped it but
# broke restart, which core/kernel.py uses), and ipykernel 7 is the kernel half.
# Below them the `python` tool still runs, but its sockets are plaintext.
jupyter_client>=8.9.1    # `python` — stateful IPython kernel (core/kernel.py)
ipykernel>=7.0           # the kernel that jupyter_client launches
pexpect>=4.9             # `interactive_run` — commands that prompt (core/processes.py)
ruamel.yaml>=0.18        # `config_edit` — YAML round-trip, comments preserved
tomlkit>=0.12            # `config_edit` — TOML round-trip, comments preserved
jsonpath-ng>=1.6         # `config_edit` — required for its '$.…' JSONPath queries
duckdb>=0.10             # `sql_query` — SQL over CSV/Parquet/JSON (core/data.py)
send2trash>=1.8          # `trash` — recoverable deletes (core/files.py)
pyautogui>=0.9.54        # `computer` — screen capture + mouse/keyboard (core/computer.py)
pillow>=10.0             # `computer` — screenshot encode/resize; also pyautogui's backend
dbus-next>=0.2.3         # `computer` on Wayland (portal remote control, core/wayland_input.py)
                         # and `desktop_window` (KWin scripting, core/desktop_window.py)
httpx2                   # `text_embeddings`/`vision_query` (core/text_embeddings.py,
                         # core/vision.py) and mcp_client.py's http transport. Also
                         # pulled in by `anthropic` and `mcp`; listed because these
                         # modules import it directly.
piper-tts                # `speak` — local neural TTS (core/speak.py), invoked
                         # as `python3 -m piper` (its CLI entry point). **Not**
                         # `sudo apt install piper` — that installs an unrelated
                         # GTK app for configuring gaming mice, same name by
                         # coincidence (see CLAUDE.md). Playback itself needs no
                         # pip package: it shells out to PipeWire's `paplay`.
faster-whisper           # `listen` — CPU/CTranslate2 speech-to-text
                         # (core/listen.py), called in-process
                         # (`WhisperModel(...).transcribe(...)`), no CLI entry
                         # point of its own. Capture itself goes through the
                         # system `parecord` binary (not pip); this package only
                         # does the transcription step.
ruff>=0.16               # linter; `pyproject.toml`'s `[tool.ruff.lint]` tunes it. `ruff check .`
                         # should come back clean; see CLAUDE.md.
mypy>=2.0                # type checker; `pyproject.toml`'s `[tool.mypy]` tunes it. A CI gate,
                         # so it is a real dependency. `mypy .` should come back clean.
