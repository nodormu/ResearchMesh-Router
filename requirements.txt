# Runtime dependencies for ResearchMesh-Router.
#   pip install -r requirements.txt
#
# The first four are the router proper. Everything under "local tools" below
# backs the tool modules merged in from ResearchMesh — the router now executes
# work itself as well as delegating it, so it needs the same backings that
# repo does, plus the same apt packages (libreoffice, pandoc, python3-tk,
# scrot, xvfb) and `playwright install chromium`.
#
# Each of those is imported *inside* its tool's `execute`, so a missing one
# costs you that tool and an install hint, not the app.

anthropic>=1,<2          # upgraded from 0.116.0 -> 1.8.0 (2026); no upper bound left it drifting
                         # silently past the 1.0.0 breaking-change boundary (httpx2, removed
                         # temperature/top_p/top_k from create(), client-side compaction_control
                         # removed) -- none of that affects this codebase (grepped clean), but
                         # pin explicitly so a fresh install matches what's actually been
                         # verified. Re-verify (ruff/mypy/smoke_test.py/test_bash_session.py/
                         # test_process_reaper.py/test_listen.py + a live call) before raising
                         # the ceiling to <3 whenever 2.0 ships.
mcp[cli]>=2.0,<3         # MCP client + Streamable HTTP transport
# The floor is 2.0, not a preference: mcp 2.0 is a breaking major and this code
# uses the new API — `streamable_http_client` in mcp_client.py (renamed from
# `streamablehttp_client`, and headers now arrive via an httpx2 client rather
# than a `headers=` argument), `read_timeout_seconds` as a plain float on
# ClientSession (it was a timedelta in 1.x), and snake_case model fields in
# core/tools.py (`input_schema`, `is_error`, `mime_type`). None of it runs on
# 1.x. The camelCase spellings survive as serialization *aliases*, so
# constructing a model still works either way while attribute reads break —
# which is why a 1.x/2.x mismatch in core/tools.py is a runtime AttributeError
# that no import check catches, and why `mypy .` is a configured gate here.
# The `<3` matches what claude-agent-sdk declares, so the two can share a venv.
prompt-toolkit>=3.0.51   # interactive CLI REPL
pydantic>=2.0            # imported directly in mcp_client.py; also required by mcp/anthropic

# httpx2 is used directly by mcp_client.py (for the long read timeout a worker
# needs) but is deliberately not listed: it arrives as a dependency of mcp, and
# pinning it here would let this file disagree with whatever mcp resolved to.
# It is imported lazily inside the http transport, so if that ever stops being
# true, only http workers break — stdio workers keep running.

# --- local tools ---------------------------------------------------------
# Backings for the tool modules in core/. Same set and same versions as
# ResearchMesh, so the two stay comparable.
playwright>=1.40.0       # `browser_*` — headless browser (core/browser.py)

# 8.9.1 is KernelManager's `transport_encryption` trait (8.9.0 shipped it but
# broke restart, which core/kernel.py uses), and ipykernel 7 is the kernel half.
# Below them the `python` tool still runs, but its sockets are plaintext.
jupyter_client>=8.9.1    # `python` — stateful IPython kernel (core/kernel.py)
ipykernel>=7.0           # the kernel that jupyter_client launches
pexpect>=4.9             # `interactive_run` — commands that prompt (core/processes.py)
ruamel.yaml>=0.18        # `config_edit` — YAML round-trip, comments preserved

jsonpath-ng>=1.6         # `config_edit` — optional, only for '$.…' JSONPath queries
duckdb>=0.10             # `sql_query` — SQL over CSV/Parquet/JSON (core/data.py)
send2trash>=1.8          # `trash` — recoverable deletes (core/files.py)
pyautogui>=0.9.54        # `computer` — screen capture + mouse/keyboard (core/computer.py)
pillow>=10.0             # `computer` — screenshot encode/resize; also pyautogui's backend
httpx>=0.24              # `text_embeddings`/`vision_query` — calls your own
                         # embedding/vision server (core/text_embeddings.py,

                         # -- verified live, anthropic 1.8.0's own metadata no

                         # 0.116.0 which did) -- an independent, explicit
                         # requirement here now, not a free ride.
piper-tts                # `speak` — local neural TTS (core/speak.py), invoked

faster-whisper           # `listen` — CPU/CTranslate2 speech-to-text

                         # (`WhisperModel(...).transcribe(...)`), no CLI entry
                         # point of its own. Capture itself goes through the
                         # system `parecord` binary (not pip); this package only
                         # does the transcription step.
